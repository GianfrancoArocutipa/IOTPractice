5. Puedes hacer clic en el sensor LDR durante la simulación para variar la luz (lux)

   y se apagan cuando sube (mucha luz).

--------------------------------------------------------------------------------
ARCHIVO: sketch.ino
--------------------------------------------------------------------------------
int sensorPin = A0; // LDR
int relePin = 9;    // relé
int ledPin = 8;     // LED
int umbral = 750;   // Valor umbral para activar el relé y el LED

void setup() {
  pinMode(relePin, OUTPUT); // pin del relé como salida
  pinMode(ledPin, OUTPUT);  // pin del LED como salida
  Serial.begin(9600);       // Iniciar comunicacion serial
}

void loop() {
  int valorLDR = analogRead(sensorPin); // Leer valor del LDR
  Serial.println(valorLDR);             // Visualizar en Monitor Serial

  // Si el valor del LDR es menor que el umbral, activar relé y LED
  if (valorLDR < umbral) {
    digitalWrite(relePin, HIGH); // Activar el relé
    digitalWrite(ledPin, HIGH);  // Encender el LED
  } else {
    digitalWrite(relePin, LOW);  // Desactivar el relé
    digitalWrite(ledPin, LOW);   // Apagar el LED
  }

  delay(1000); // Pequeña pausa para evitar lecturas continuas
}
