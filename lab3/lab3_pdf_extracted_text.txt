Introducción 
 
 
ESP32 es la denominación de una familia de chips SoC de bajo costo y consumo de 
energía, con tecnología Wi-Fi y Bluetooth de modo dual integrada. El ESP32 emplea un 
microprocesador Tensilica Xtensa LX6 en sus variantes de simple y doble núcleo e 
incluye interruptores de antena, balun de radiofrecuencia, amplificador de potencia, 
amplificador receptor de bajo ruido, filtros, y módulos de administración de energía. El 
ESP32 fue creado y desarrollado por Espressif Systems y es fabricado por TSMC 
utilizando su proceso de 40 nm. Es un sucesor de otro SoC, el ESP8266. 
ESP 32 CARACTERISTICAS 
• Procesadores: 
• Procesador principal: microprocesador Tensilica Xtensa LX6 de 32 bits 
• Núcleos: 2 o 1 (según variación). Todos los chips de la serie ESP32 son de doble 
núcleo, excepto el ESP32 -S0WD, que es de un solo núcleo. Frecuencia de reloj: 
hasta 240 MHz 
• Rendimiento: hasta 600 DMIPS 
• Coprocesador de potencia ultrabaja: le permite realizar conversiones ADC, cálculos 
y umbrales de nivel mientras está en sueño profundo. 
• Conectividad inalámbrica: 
• Wi-Fi: 802.11 b / g / n / e / i (802.11n @ 2.4 GHz hasta 150 Mbit / s) 
• Bluetooth: v4.2 BR / EDR y Bluetooth de baja energía (BLE)

/* 
 *      -----TEST SENSORES DHT22  
*/ 
#include "DHT.h"             // incluimos el módulo de temperatura 
#define pin1 13 
 
int pin2=14; 
int Volt=0; 
DHT dht(pin1, DHT22);    //El sensor blanco (DHT11 es el azul) 
 
void setup() { 
  Serial.begin(115200); 
  Serial.println("Test de sensores:"); 
  dht.begin(); 
} 
void loop() { 
  delay(500); 
  leerdht(); 
  delay(500); 
  leerpot(); 
} 
void leerdht() { 
  float h = dht.readHumidity(); //Leemos la Humedad 
  float tc = dht.readTemperature(); //Leemos la temperatura en Celsius 
  float tf = dht.readTemperature(true);//Leemos la temperatura en Fahrenheit 
  float  st= dht.computeHeatIndex(tc,h); //Sensación Termica 
  Serial.println("Temperatura: " + String(tc)+ " °C "); 
  Serial.println("Humedad: " + String(h) ); 
  Serial.println("Temperatura: " + String(tf)+ " °F "); 
  Serial.println("Sensación Térmica: " + String(st) + "\n"); 
  Serial.println("--------------------------------------------"); 
  delay (2000); 
} 
void leerpot() { 
  int potValor = analogRead(pin2); 
  Serial.println("Volaje de entrada:" + String(potValor)+ "  binario" 
);  //Envío del valor al puerto serie 
  Serial.println("--------------------------------------------"); 
  delay(500); 
}

// Leer temperatura ---------------------------------------------------- 
void leerdht() { 
 
    float tc = dht.readTemperature(); 
    Serial.println("Temperatura : " + String(tc) + " °C"); 
   temperatura.publish(tc);

//Leer potenciómetro ------------------------------------------- 
void leerpot() { 
  int lectura = analogRead(pin2); 
  // Conversión a voltaje (ESP32 ADC de 12 bits) 
  float vol = lectura * 3.3 / 4095.0; 
  Serial.println("ADC      : " + String(lectura)); 
  Serial.println("Voltaje  : " + String(vol) + " V"); 
  voltaje.publish(vol); 
} 
 
//conectar al MQTT -------------------------------------------- 
void conectarMQTT() { 
  int8_t ret; 
  if (mqtt.connected()) { 
    return; 
  } 
 
  Serial.print("Conectando a Adafruit IO... "); 
  while ((ret = mqtt.connect()) != 0) { 
    Serial.println(mqtt.connectErrorString(ret)); 
    Serial.println("Reintentando en 5 segundos..."); 
    mqtt.disconnect(); 
    delay(5000); 
  } 
  Serial.println("Conectado"); 
}
